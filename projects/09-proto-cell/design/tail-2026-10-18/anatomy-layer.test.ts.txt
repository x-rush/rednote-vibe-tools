import { describe, expect, it, vi } from 'vitest'
import { createAnatomyLayer } from './anatomy-layer'
import * as anatomy from './late-anatomy'

const palette = { membrane: '#ff9b74', cytoplasm: '#181228', core: '#ffcf70', organ: '#ff5b87', glow: '#ff596f' }
const makeContext = () => new Proxy({ globalAlpha: 1, globalCompositeOperation: 'source-over', drawImage: vi.fn() }, { get: (target, name) => Reflect.get(target, name) ?? (() => {}) }) as unknown as CanvasRenderingContext2D
const makeSurface = () => ({ width: 0, height: 0, getContext: () => makeContext() }) as unknown as HTMLCanvasElement

describe('dense late anatomy layers', () => {
 it('shares only identical same-frame anatomy and redraws every changed animation frame', () => {
  const surfaces: HTMLCanvasElement[] = [], vectors=vi.spyOn(anatomy, 'drawLateCreatureAnatomy'), context=makeContext()
  const layer=createAnatomyLayer(()=>{const canvas=makeSurface();surfaces.push(canvas);return canvas})
  try {
   layer.beginFrame()
   for(let n=0;n<120;n++)expect(layer.draw(context,'predator-chamber-maw',32,1900,palette,false,'balanced')).toBe(true)
   expect(vectors).toHaveBeenCalledTimes(1)
   expect(context.drawImage).toHaveBeenCalledTimes(120)
   layer.draw(context,'predator-chamber-maw',32,1900,palette,true,'balanced')
   layer.draw(context,'predator-chamber-maw',32,1900,{...palette,core:'#ffffff'},false,'balanced')
   expect(vectors).toHaveBeenCalledTimes(3)
   layer.beginFrame();layer.draw(context,'predator-chamber-maw',32,1916,palette,false,'balanced')
   expect(vectors).toHaveBeenCalledTimes(4)
   expect(surfaces).toHaveLength(3)
   layer.destroy();expect(surfaces.every(s=>s.width===0&&s.height===0)).toBe(true)
  } finally { vectors.mockRestore() }
 })
 it('caps temporary canvas memory and falls back for unsupported or opacity-sensitive draws', () => {
  const surfaces: HTMLCanvasElement[]=[],layer=createAnatomyLayer(()=>{const c=makeSurface();surfaces.push(c);return c}),context=makeContext()
  layer.beginFrame()
  for(let n=0;n<40;n++)layer.draw(context,'predator-chamber-maw',30+n,1900,palette,false,'balanced')
  expect(surfaces.length).toBeLessThanOrEqual(8)
  expect(surfaces.reduce((sum,s)=>sum+s.width*s.height,0)).toBeLessThanOrEqual(1_048_576)
  context.globalAlpha=.38;expect(layer.draw(context,'predator-chamber-maw',32,1900,palette,false,'balanced')).toBe(false)
  context.globalAlpha=1;expect(layer.draw(context,'other-species',32,1900,palette,false,'balanced')).toBe(false)
  expect(layer.draw(context,'predator-chamber-maw',300,1900,palette,false,'high')).toBe(false)
  layer.destroy()
 })
})

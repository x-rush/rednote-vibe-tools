import { describe, expect, it, vi } from 'vitest'
import { createSimpleCellLayer } from './simple-layer'
import type { CellVisualProfile } from './cell'
const profile: CellVisualProfile={silhouette:'colony',appendages:'spines',palette:{membrane:'#ffd184',cytoplasm:'rgb(128 28 91 / 38%)',core:'#ffa645',organ:'#cb903e',glow:'#ff596f'}}
const context=()=>new Proxy({globalAlpha:1,globalCompositeOperation:'source-over',drawImage:vi.fn(),ellipse:vi.fn()},{get:(target,name)=>Reflect.get(target,name)??(()=>{})}) as unknown as CanvasRenderingContext2D

describe('dense simple cell surface reuse',()=>{
 it('reuses same-frame shared silhouettes with less than 1/64px radius error and refreshes animation',()=>{
  const surfaces:HTMLCanvasElement[]=[], targets:CanvasRenderingContext2D[]=[],layer=createSimpleCellLayer(()=>{const target=context();targets.push(target);const c={width:0,height:0,getContext:()=>target} as unknown as HTMLCanvasElement;surfaces.push(c);return c}),screen=context()
  layer.beginFrame()
  for(let n=0;n<100;n++)expect(layer.draw(screen,profile,32,32+n*.00001,1900)).toBe(true)
  expect(surfaces).toHaveLength(1)
  expect(screen.drawImage).toHaveBeenCalledTimes(100)
  expect(targets[0]!.ellipse).toHaveBeenCalledTimes(1)
  layer.beginFrame();layer.draw(screen,profile,32,32,1916)
  expect(surfaces).toHaveLength(1);expect(targets[0]!.ellipse).toHaveBeenCalledTimes(2)
  layer.destroy();expect(surfaces[0]!.width).toBe(0)
 })
 it('bounds allocation and keeps ID-dependent shapes and translucent sessions on the vector path',()=>{
  const surfaces:HTMLCanvasElement[]=[],layer=createSimpleCellLayer(()=>{const c={width:0,height:0,getContext:()=>context()} as unknown as HTMLCanvasElement;surfaces.push(c);return c}),screen=context()
  layer.beginFrame();for(let n=0;n<50;n++)layer.draw(screen,profile,30+n,30+n,1900)
  expect(surfaces.length).toBeLessThanOrEqual(16);expect(surfaces.reduce((s,c)=>s+c.width*c.height,0)).toBeLessThanOrEqual(1_048_576)
  expect(layer.draw(screen,{...profile,silhouette:'hunter'},32,32,1900)).toBe(false)
  screen.globalAlpha=.38;expect(layer.draw(screen,profile,32,32,1900)).toBe(false)
  layer.destroy()
 })
})
